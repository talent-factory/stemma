\startuniqueMPgraphic{sealframe}
  fill OverlayBox withcolor \MPcolor{sealground} ;
  draw OverlayBox enlarged -1.6cm
    withpen pencircle scaled 1pt withcolor \MPcolor{sealink} ;
\stopuniqueMPgraphic
\defineoverlay[sealframe][\useMPgraphic{sealframe}]

% Status: a sash across the top right corner; it disappears when status is empty.
\startuniqueMPgraphic{statussash}
  pair c ; c := (OverlayWidth-2.6cm, OverlayHeight-2.6cm) ;
  fill ((-4cm,-.55cm)--(4cm,-.55cm)--(4cm,.55cm)--(-4cm,.55cm)--cycle)
    rotated -45 shifted c withcolor \MPcolor{sealcolor} ;
  draw textext("\ss\bf\tfa\color[sealground]{\getvariable{document}{status}}")
    rotated -45 shifted c ;
  clip currentpicture to OverlayBox ;
  setbounds currentpicture to OverlayBox ;
\stopuniqueMPgraphic
\defineoverlay[statussash][\useMPgraphic{statussash}]

\setupbackgrounds[page][background=sealframe]
\doifsomething{\getvariable{document}{status}}
